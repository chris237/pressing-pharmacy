INTÉGRATION DANS LE SERVEUR

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

À ajouter dans server.js:

  const helmet = require('helmet');
  const rateLimit = require('express-rate-limit');
  const auth = require('./auth');
  const authRoutes = require('./auth-routes');

  // Sécurité
  app.use(helmet());

  // Rate limiting
  const limiter = rateLimit({
    windowMs: 15 * 60 * 1000,
    max: 100
  });
  app.use(limiter);

  // Routes d'authentification
  app.use('/api', authRoutes);

  // Initialiser les rôles par défaut
  auth.initializeDefaultRoles();

DÉPLOIEMENT

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Installer nouvelles dépendances:
   npm install

   Nouvelles packages:
   - jsonwebtoken (JWT)
   - bcryptjs (mots de passe)
   - json2csv (export)
   - helmet (sécurité)
   - express-rate-limit (rate limit)
   - express-validator (validation)

2. Démarrer le serveur:
   npm start

3. Ajouter les routes dans server.js:
   - Importer auth.js et auth-routes.js
   - Ajouter les middlewares
   - Appeler initializeDefaultRoles()

4. Accéder:
   http://localhost:3000/public/auth.html

5. Créer premier admin (optionnel):
   Via POST /api/admin/users avec token admin
